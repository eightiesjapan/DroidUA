cmake_minimum_required(VERSION 3.18)
project(OpenUAAndroid LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(UA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(ANDROID_WORK "${UA_ROOT}/../Android")
set(DEPS "${ANDROID_WORK}/Dependencies")
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${ANDROID_WORK}/SDL2" "${CMAKE_BINARY_DIR}/SDL2")

# OpenUA includes SDL headers using the desktop-style SDL2/ prefix.
set(COMPAT "${CMAKE_BINARY_DIR}/compat")
file(MAKE_DIRECTORY "${COMPAT}/SDL2")
file(GLOB SDL_HEADERS "${ANDROID_WORK}/SDL2/include/*.h")
foreach(header IN LISTS SDL_HEADERS)
    configure_file("${header}" "${COMPAT}/SDL2/" COPYONLY)
endforeach()
configure_file("${DEPS}/SDL_image/include/SDL_image.h" "${COMPAT}/SDL2/SDL_image.h" COPYONLY)
configure_file("${DEPS}/SDL_ttf/SDL_ttf.h" "${COMPAT}/SDL2/SDL_ttf.h" COPYONLY)
configure_file("${DEPS}/SDL_net/SDL_net.h" "${COMPAT}/SDL2/SDL_net.h" COPYONLY)

set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(SDL2IMAGE_AVIF OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_TIF OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_WEBP OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_JXL OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_VENDORED OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_BACKEND_STB ON CACHE BOOL "" FORCE)
set(SDL2IMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/SDL_image" "${CMAKE_BINARY_DIR}/SDL_image")

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(feature ZLIB BZIP2 PNG BROTLI HARFBUZZ)
    set(FT_DISABLE_${feature} ON CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${DEPS}/freetype" "${CMAKE_BINARY_DIR}/freetype")
add_library(SDL2_ttf STATIC "${DEPS}/SDL_ttf/SDL_ttf.c")
target_include_directories(SDL2_ttf PUBLIC "${DEPS}/SDL_ttf")
target_link_libraries(SDL2_ttf PUBLIC SDL2 freetype)
add_library(SDL2_net STATIC "${DEPS}/SDL_net/SDLnet.c" "${DEPS}/SDL_net/SDLnetTCP.c" "${DEPS}/SDL_net/SDLnetUDP.c" "${DEPS}/SDL_net/SDLnetselect.c")
target_include_directories(SDL2_net PUBLIC "${DEPS}/SDL_net")
target_link_libraries(SDL2_net PUBLIC SDL2)

set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(INSTALL_DOCS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/ogg" "${CMAKE_BINARY_DIR}/ogg")
file(READ "${DEPS}/vorbis/lib/CMakeLists.txt" VORBIS_CMAKE)
string(REGEX MATCH "set\\(VORBIS_SOURCES([^)]*)\\)" _ "${VORBIS_CMAKE}")
string(REGEX REPLACE "[ \t\r\n]+" ";" VORBIS_SOURCES "${CMAKE_MATCH_1}")
list(FILTER VORBIS_SOURCES EXCLUDE REGEX "^$")
list(TRANSFORM VORBIS_SOURCES PREPEND "${DEPS}/vorbis/lib/")
add_library(vorbis STATIC ${VORBIS_SOURCES} "${DEPS}/vorbis/lib/vorbisfile.c")
target_include_directories(vorbis PUBLIC "${DEPS}/vorbis/include" PRIVATE "${DEPS}/vorbis/lib")
target_link_libraries(vorbis PUBLIC ogg m)

file(GLOB LUA_SOURCES "${DEPS}/lua/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|onelua|ltests)\\.c$")
add_library(lua STATIC ${LUA_SOURCES})
target_include_directories(lua PUBLIC "${DEPS}/lua")
target_compile_definitions(lua PRIVATE LUA_USE_LINUX)
target_link_libraries(lua PUBLIC dl m)

set(LIBTYPE STATIC CACHE STRING "" FORCE)
set(ALSOFT_UTILS OFF CACHE BOOL "" FORCE)
set(ALSOFT_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ALSOFT_TESTS OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/openal-soft" "${CMAKE_BINARY_DIR}/openal")
# Newer NDK headers require an explicit standard allocation declaration.
target_compile_options(alsoft.fmt PRIVATE -include cstdlib)
target_compile_options(OpenAL PRIVATE -include cstdlib)

set(NOX11 ON CACHE BOOL "" FORCE)
set(NOEGL ON CACHE BOOL "" FORCE)
set(STATICLIB OFF CACHE BOOL "" FORCE)
set(NO_INIT_CONSTRUCTOR ON CACHE BOOL "" FORCE)
set(USE_ANDROID_LOG ON CACHE BOOL "" FORCE)
set(DEFAULT_ES 2 CACHE STRING "" FORCE)
add_subdirectory("${DEPS}/gl4es" "${CMAKE_BINARY_DIR}/gl4es")
set_target_properties(GL PROPERTIES OUTPUT_NAME gl4es SUFFIX ".so" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}")
target_link_libraries(GL dl m)

# Reuse the upstream source list without importing desktop package discovery.
file(READ "${UA_ROOT}/src/CMakeLists.txt" UA_CMAKE)
string(REGEX MATCH "set\\(SOURCES([^)]*)\\)" _ "${UA_CMAKE}")
string(REGEX REPLACE "[ \t\r\n]+" ";" UA_SOURCES "${CMAKE_MATCH_1}")
list(FILTER UA_SOURCES EXCLUDE REGEX "^$|^system/movie\\.cpp$")
list(TRANSFORM UA_SOURCES PREPEND "${UA_ROOT}/src/")
add_library(main SHARED ${UA_SOURCES} movie_stub.cpp android_controls.cpp "${UA_ROOT}/src/system/android_display.cpp")
set_target_properties(main PROPERTIES CXX_STANDARD 14)
target_compile_definitions(main PRIVATE UA_ANDROID_PROTOTYPE=1)
target_include_directories(main PRIVATE "${COMPAT}" "${DEPS}/gl4es/include" "${DEPS}/openal-soft/include" "${UA_ROOT}/src")
target_link_libraries(main PRIVATE SDL2 SDL2_image SDL2_ttf SDL2_net OpenAL vorbis lua GL log android)
